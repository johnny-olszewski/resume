% \name{Name} — centered name with horizontal rule
\newcommand{\name}[1]{%
  \begin{center}
    \Large\textbf{#1}
  \end{center}
  \vspace{-0.4em}
  \hrule height 0.4pt
  \vspace{0.2em}
}

% \contact{email}{phone}{location}{linkedin} — single-line contact info
\newcommand{\contact}[4]{%
  \begin{center}
    \small #1 \textbar{} #2 \textbar{} #3 \textbar{} #4
  \end{center}
  \vspace{-0.35em}
  \hrule height 0.4pt
  \vspace{0.2em}
}

% \summary{text} — professional summary block
\newcommand{\summary}[1]{%
  \vspace{-0.2em}
  \begin{center}
    \footnotesize\textit{#1}
  \end{center}
}

% \role{Company}{Location}{Title}{Date Range} — job entry header
\newcommand{\role}[4]{%
  \vspace{0.4em}

}

% \project{Name}{Tech Stack} — project entry header
\newcommand{\project}[2]{%

  #3
}
